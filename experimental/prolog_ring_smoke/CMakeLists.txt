cmake_minimum_required(VERSION 3.16)
set(SFA_SDK_ROOT "$ENV{ASCEND_HOME_PATH}" CACHE PATH "Existing matched toolkit; never switch installations")
find_package(ASC REQUIRED)
project(qsfa_prolog_ring_smoke LANGUAGES ASC CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Smoke build type" FORCE)
endif()
if(NOT CMAKE_ASC_ARCHITECTURES STREQUAL "dav-3510")
    message(FATAL_ERROR "Use CMAKE_ASC_ARCHITECTURES=dav-3510 for this isolated A5 smoke")
endif()
set(SFA_ASC_ROOT "${SFA_SDK_ROOT}/aarch64-linux/asc" CACHE PATH "ASC headers in the selected toolkit")
get_filename_component(sdk_real "${SFA_SDK_ROOT}" REALPATH)
foreach(path "${CMAKE_ASC_COMPILER}" "${SFA_ASC_ROOT}")
    get_filename_component(path_real "${path}" REALPATH)
    string(FIND "${path_real}/" "${sdk_real}/" within_sdk)
    if(NOT within_sdk EQUAL 0)
        message(FATAL_ERROR "Compiler and headers must belong to SFA_SDK_ROOT: ${path}")
    endif()
endforeach()
find_path(ACL_INCLUDE acl/acl.h PATHS "${SFA_SDK_ROOT}/aarch64-linux/include" "${SFA_SDK_ROOT}/include" NO_DEFAULT_PATH)
find_library(ACL_LIBRARY ascendcl PATHS "${SFA_SDK_ROOT}/aarch64-linux/lib64" "${SFA_SDK_ROOT}/lib64" NO_DEFAULT_PATH)
if(NOT ACL_INCLUDE OR NOT ACL_LIBRARY)
    message(FATAL_ERROR "Matched ACL headers/library missing in SFA_SDK_ROOT")
endif()
add_library(qsfa_prolog_ring_smoke SHARED prolog_ring_smoke.asc)
set_target_properties(qsfa_prolog_ring_smoke PROPERTIES POSITION_INDEPENDENT_CODE ON UNITY_BUILD OFF)
target_compile_options(qsfa_prolog_ring_smoke PRIVATE $<$<COMPILE_LANGUAGE:ASC>:--npu-arch=dav-3510> $<$<COMPILE_LANGUAGE:ASC>:-O2>)
target_include_directories(qsfa_prolog_ring_smoke PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${ACL_INCLUDE}")
foreach(suffix "" "/include" "/include/basic_api" "/include/adv_api" "/include/interface" "/include/tiling" "/include/utils" "/include/comm_api" "/impl" "/impl/basic_api")
    if(EXISTS "${SFA_ASC_ROOT}${suffix}")
        target_include_directories(qsfa_prolog_ring_smoke PRIVATE "${SFA_ASC_ROOT}${suffix}")
    endif()
endforeach()
target_link_libraries(qsfa_prolog_ring_smoke PRIVATE m "${ACL_LIBRARY}" ascendc_runtime)
message(STATUS "Isolated ring-writer smoke only; no install target, vendor edits, Prolog or C4 validation")
