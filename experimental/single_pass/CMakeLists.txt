cmake_minimum_required(VERSION 3.16)
set(SFA_SDK_ROOT "$ENV{ASCEND_HOME_PATH}" CACHE PATH "Existing matched CANN toolkit; never switch installations")
find_package(ASC REQUIRED)
project(qsfa_recent128_single_pass LANGUAGES ASC CXX)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Prototype build type" FORCE)
endif()
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(BUNDLE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(C4_ROOT "${BUNDLE_ROOT}/native_c4_source/source/ascendc/kv_quant_sparse_flash_attention")
set(C4_ARCH "${C4_ROOT}/arch35")
set(SFA_ASC_ROOT "${SFA_SDK_ROOT}/aarch64-linux/asc" CACHE PATH "ASC root within the same toolkit")
foreach(required "${C4_ARCH}/common/op_kernel/buffer.h" "${C4_ARCH}/kv_quant_sparse_flash_attention_service_cube_mla.h")
    if(NOT EXISTS "${required}")
        message(FATAL_ERROR "Matched C4 source handoff incomplete: ${required}")
    endif()
endforeach()
if(NOT CMAKE_ASC_ARCHITECTURES STREQUAL "dav-3510")
    message(FATAL_ERROR "Set CMAKE_ASC_ARCHITECTURES=dav-3510 for the A5 prototype")
endif()
get_filename_component(sdk_real "${SFA_SDK_ROOT}" REALPATH)
foreach(path "${CMAKE_ASC_COMPILER}" "${SFA_ASC_ROOT}")
    get_filename_component(path_real "${path}" REALPATH)
    string(FIND "${path_real}/" "${sdk_real}/" within_sdk)
    if(NOT within_sdk EQUAL 0)
        message(FATAL_ERROR "Compiler/include must belong to the selected toolkit: ${path}")
    endif()
endforeach()
find_path(ACL_INCLUDE acl/acl.h PATHS "${SFA_SDK_ROOT}/aarch64-linux/include" "${SFA_SDK_ROOT}/include" NO_DEFAULT_PATH)
find_library(ACL_LIBRARY ascendcl PATHS "${SFA_SDK_ROOT}/aarch64-linux/lib64" "${SFA_SDK_ROOT}/lib64" NO_DEFAULT_PATH)
if(NOT ACL_INCLUDE OR NOT ACL_LIBRARY)
    message(FATAL_ERROR "No matched ACL headers/library in SFA_SDK_ROOT")
endif()
add_library(qsfa_recent128_direct SHARED single_pass_wrapper.asc original_c4_wrapper.asc routed_single_pass_wrapper.asc)
set_target_properties(qsfa_recent128_direct PROPERTIES POSITION_INDEPENDENT_CODE ON UNITY_BUILD OFF)
target_compile_options(qsfa_recent128_direct PRIVATE
    $<$<COMPILE_LANGUAGE:ASC>:--npu-arch=dav-3510>
    $<$<COMPILE_LANGUAGE:ASC>:-O2>)
target_include_directories(qsfa_recent128_direct PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${C4_ARCH}" "${C4_ROOT}"
    "${C4_ARCH}/common/op_kernel" "${ACL_INCLUDE}" "${BUNDLE_ROOT}/deps/ascendc_common")
foreach(suffix "" "/include" "/include/basic_api" "/include/adv_api" "/include/interface" "/include/tiling"
        "/include/utils" "/include/comm_api" "/impl" "/impl/basic_api")
    if(EXISTS "${SFA_ASC_ROOT}${suffix}")
        target_include_directories(qsfa_recent128_direct PRIVATE "${SFA_ASC_ROOT}${suffix}")
    endif()
endforeach()
target_link_libraries(qsfa_recent128_direct PRIVATE m "${ACL_LIBRARY}" ascendc_runtime)
message(STATUS "Standalone ABI1; separate overlay/routed entries with unchanged native C4 control; no device validation claimed")
