cmake_minimum_required(VERSION 3.16)
find_package(ASC REQUIRED)

project(sfa_launch LANGUAGES ASC CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

set(CMAKE_ASC_ARCHITECTURES "dav-3510" CACHE STRING "A5")
set(ASC_ROOT "/usr/local/Ascend/cann-9.2.1/aarch64-linux/asc")
set(PKG_INC "/usr/local/Ascend/cann-9.2.1/opp/vendors/custom_cot/op_impl/ai_core/tbe/custom_cot_impl/ascendc/sparse_flash_attention")
set(COMMON_ASCENDC "/usr/local/Ascend/cann-9.2.1/opp/vendors/custom_cot/op_impl/ai_core/tbe/custom_cot_impl/ascendc")
set(SFA_BUNDLE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/..")

# Device kernel (ASC) + Host launcher (C++)
add_executable(sfa_launch
    sfa_direct_launch.cpp
    sfa_standalone_entry.asc
)

target_include_directories(sfa_launch PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}
    ${SFA_BUNDLE_ROOT}/original
    ${SFA_BUNDLE_ROOT}/original/arch35
    ${SFA_BUNDLE_ROOT}/deps/sparse_flash_attention/arch35
    ${ASC_ROOT}
    ${ASC_ROOT}/include
    ${ASC_ROOT}/include/basic_api
    ${ASC_ROOT}/include/adv_api
    ${ASC_ROOT}/include/interface
    ${ASC_ROOT}/impl
    ${ASC_ROOT}/impl/basic_api
    ${PKG_INC}
    ${PKG_INC}/arch35
    ${COMMON_ASCENDC}
)

# ACL runtime for host code
find_library(ACL_LIB ascendcl PATHS ${ASC_ROOT}/../lib64 NO_DEFAULT_PATH)
target_link_libraries(sfa_launch PRIVATE
    ${ACL_LIB}
    ascendc_runtime
)

target_compile_options(sfa_launch PRIVATE
    $<$<COMPILE_LANGUAGE:ASC>:--npu-arch=${CMAKE_ASC_ARCHITECTURES}>
)
